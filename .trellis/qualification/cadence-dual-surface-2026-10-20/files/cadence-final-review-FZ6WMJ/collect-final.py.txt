import pathlib,json,hashlib,os,time,shutil,subprocess
r=pathlib.Path(__file__).parent; repo=pathlib.Path('/home/mirako/abel/cadence')
h=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
ref=lambda p:{'path':str(p),'bytes':p.stat().st_size,'sha256':h(p)}
names=['verify-repaired','real-pack-final','real-extract-final','extracted-browser-final','extracted-native-final','extracted-operations-final']
receipts={}; browser_roots=[]
for name in names:
 c=json.loads((r/(name+'.json')).read_text()); assert c['originalWait'] and c['exit']==0
 assert h(pathlib.Path(c['log']))==c['logSha256']
 start=c['started']; end=start+c['elapsedSeconds']+1
 paths=[]
 for pattern in ['*/receipt.json','*/pty-receipt.json','*/owner-receipt.json','*/recovery-restart-receipt.json','*/distribution-diagnostics.json','*/lifetime-receipt.json','*/sdk-receipt.json','*/check-receipt.json','accepting-*/mixed-pi/pty-receipt.json','cadence-terminal-*/reattach/pty-receipt.json']:
  paths += [p for p in r.glob(pattern) if start <= p.stat().st_mtime <= end and 'extracted' not in p.parts]
 for pattern in ['cdb-*/receipt.json','cdn-*/receipt.json']:
  for p in r.parent.glob(pattern):
   if start<=p.stat().st_mtime<=end:paths.append(p);browser_roots.append(str(p.parent))
 rows=[]
 for p in sorted(set(paths)):
  v=json.loads(p.read_text());row=ref(p)
  for k in ['passed','mode','packageRoot','encoding','original','originalClose','originalBrowserClose','stopped','restartedCloses','forced','code','signal','waitpidReturned','exitCode','failure','contacts','maxActive','refresh','inference','direct','physicalHuman','paidProvider','crash','restarted','cleanRestart','replayed']:
   if k in v:row[k]=v[k]
  if p.name=='owner-receipt.json':
   assert v['code']==0 and v['signal'] is None and not v['forced']
   assert len(v['terminalReceipts'])==2
   assert all(x['waitpidReturned'] and x['exitCode']==0 and x['failure'] is None for x in v['terminalReceipts'])
   row['originalPiWaits']=[{k:x[k] for k in ['pid','waitpidReturned','exitCode','failure']} for x in v['terminalReceipts']]
  if p.parent.name.startswith(('cdb-','cdn-')):
   assert v['passed'] and v['originalBrowserClose']==[0,None]
   row.update({k:v[k] for k in ['viewports','coldRestart','proxy','nativeObserver'] if k in v})
   row['downloads']=[ref(pathlib.Path(d['path']))|{'scope':d['scope'],'format':d['format']} for d in v.get('downloads',[])]
  if p.parent.name.startswith('owner-proxy-'):
   assert v['original']==[0,None] and v['forced']==False and v['refresh']==v['inference']==1 and v['direct']==0
  rows.append(row)
 receipts[name]=rows
(r/'final-receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
freeze=json.loads((r/'final-inputs-repaired.json').read_text());assert all(h(repo/n)==w['sha256'] for n,w in freeze.items())
(r/'frozen-after-matrix.json').write_text(json.dumps({'inputs':ref(r/'final-inputs-repaired.json'),'count':len(freeze),'changed':[]},indent=2)+'\n')
# Read only identity/process locations, no environment/credentials and no signals.
exclude={os.getpid(),os.getppid()};candidates=[]
for p in pathlib.Path('/proc').iterdir():
 if not p.name.isdigit() or int(p.name) in exclude:continue
 try:
  cmd=(p/'cmdline').read_bytes().replace(b'\0',b' ').decode(errors='replace');cwd=os.readlink(p/'cwd')
 except (FileNotFoundError,PermissionError,ProcessLookupError):continue
 # Shell wrappers can quote this path; record only actual native test processes.
 executable=cmd.split(' ',1)[0]
 if pathlib.Path(executable).name not in ['node','bun','python3','chrome','chrome_crashpad_handler']:continue
 if str(r) in cmd or cwd.startswith(str(r)) or any(b in cmd or cwd.startswith(b) for b in browser_roots):
  candidates.append({'pid':int(p.name),'executable':executable,'cwd':cwd})
(r/'final-process-inventory.json').write_text(json.dumps({'observedAt':time.time(),'ownedPathCandidates':candidates,'signalsSent':False,'absenceIsNotClosureEvidence':True},indent=2)+'\n')
assert not candidates,candidates
print(json.dumps({'receipts':{k:len(v) for k,v in receipts.items()},'inputCount':len(freeze),'ownedProcesses':candidates},indent=2))
